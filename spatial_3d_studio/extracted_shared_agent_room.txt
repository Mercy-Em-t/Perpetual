def generate_scene_image(prompt_text, reference_image_path=None):
   client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY"))
   
   # Prepare contents including prompt and optional reference images
   contents = [prompt_text]
   
   if reference_image_path and os.path.exists(reference_image_path):
       with open(reference_image_path, "rb") as f:
           image_bytes = f.read()
       contents.append(
           types.Part.from_bytes(
               data=image_bytes,
               mime_type="image/jpeg",
           )
       )
   # Call the model (e.g., Imagen or multimodal generation)
   response = client.models.generate_content(
       model='imagen-3.0-generate-002', # or relevant image/video model
       contents=contents,
   )
   
   return response
